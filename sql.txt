SELECT
    TB_CHAMADO.CHAMADO_ID AS ID,
    TB_CHAMADO.DATA_ABERTURA AS ABERTURA,
    TB_CHAMADO.DATA_FECHAMENTO AS FECHAMENTO,
    TB_CHAMADO.PRIORIDADE,
    TB_CHAMADO.STATUS,
    CLIENTE.NOME AS CLIENTE,
    TECNICO.NOME AS TECNICO,
    TB_CHAMADO.TITULO,
    TB_CHAMADO.OBSERVACOES
FROM TB_CHAMADO
JOIN TB_PESSOA CLIENTE
    ON TB_CHAMADO.CLIENTE_ID = CLIENTE.PESSOA_ID
JOIN TB_PESSOA TECNICO
    ON TB_CHAMADO.TECNICO_ID = TECNICO.PESSOA_ID;
